CREATE TABLE cliente (
   cliente_id       NUMBER          GENERATED BY DEFAULT AS IDENTITY,
   rut              VARCHAR2(12)    NOT NULL,
   nombre           VARCHAR2(60)    NOT NULL,
   apellido         VARCHAR2(60)    NOT NULL,
   telefono         VARCHAR2(20),
   email            VARCHAR2(100),
   direccion        VARCHAR2(150),
   comuna           VARCHAR2(60),
   fecha_registro   DATE            DEFAULT SYSDATE NOT NULL,
   CONSTRAINT pk_cliente PRIMARY KEY (cliente_id),
   CONSTRAINT uq_cliente_rut UNIQUE (rut)
);
COMMENT ON TABLE cliente IS 'Personas o empresas que solicitan reparaciones en TecnoFix';


CREATE TABLE sucursal (
   sucursal_id      NUMBER          GENERATED BY DEFAULT AS IDENTITY,
   nombre_sucursal  VARCHAR2(80)    NOT NULL,
   comuna           VARCHAR2(60)    NOT NULL,
   direccion        VARCHAR2(150)   NOT NULL,
   telefono         VARCHAR2(20),
   CONSTRAINT pk_sucursal PRIMARY KEY (sucursal_id)
);
COMMENT ON TABLE sucursal IS 'Sucursales de TecnoFix en distintas comunas de la Region Metropolitana';


CREATE TABLE especialidad (
   especialidad_id      NUMBER        GENERATED BY DEFAULT AS IDENTITY,
   nombre_especialidad  VARCHAR2(60)  NOT NULL,
   descripcion          VARCHAR2(200),
   CONSTRAINT pk_especialidad PRIMARY KEY (especialidad_id),
   CONSTRAINT uq_especialidad_nombre UNIQUE (nombre_especialidad)
);
COMMENT ON TABLE especialidad IS 'Catalogo de especialidades tecnicas (ej. notebooks, impresoras, redes)';

CREATE TABLE tecnico (
   tecnico_id           NUMBER          GENERATED BY DEFAULT AS IDENTITY,
   rut                  VARCHAR2(12)    NOT NULL,
   nombre               VARCHAR2(60)    NOT NULL,
   apellido             VARCHAR2(60)    NOT NULL,
   telefono             VARCHAR2(20),
   email                VARCHAR2(100),
   sucursal_id          NUMBER          NOT NULL,
   fecha_contratacion   DATE            DEFAULT SYSDATE NOT NULL,
   habilitado           CHAR(1)         DEFAULT 'S' NOT NULL,
   CONSTRAINT pk_tecnico PRIMARY KEY (tecnico_id),
   CONSTRAINT uq_tecnico_rut UNIQUE (rut),
   CONSTRAINT fk_tecnico_sucursal FOREIGN KEY (sucursal_id)
      REFERENCES sucursal (sucursal_id),
   CONSTRAINT ck_tecnico_habilitado CHECK (habilitado IN ('S','N'))
);
COMMENT ON TABLE tecnico IS 'Tecnicos de TecnoFix; habilitado controla la regla de asignacion de trabajos';


CREATE TABLE tecnico_especialidad (
   tecnico_id        NUMBER    NOT NULL,
   especialidad_id   NUMBER    NOT NULL,
   fecha_asignacion  DATE      DEFAULT SYSDATE NOT NULL,
   CONSTRAINT pk_tecnico_especialidad PRIMARY KEY (tecnico_id, especialidad_id),
   CONSTRAINT fk_tecesp_tecnico FOREIGN KEY (tecnico_id)
      REFERENCES tecnico (tecnico_id),
   CONSTRAINT fk_tecesp_especialidad FOREIGN KEY (especialidad_id)
      REFERENCES especialidad (especialidad_id)
);
COMMENT ON TABLE tecnico_especialidad IS 'Especialidades que posee cada tecnico (relacion N:M)';

CREATE TABLE tipo_equipo (
   tipo_equipo_id   NUMBER        GENERATED BY DEFAULT AS IDENTITY,
   nombre_tipo      VARCHAR2(60)  NOT NULL,
   CONSTRAINT pk_tipo_equipo PRIMARY KEY (tipo_equipo_id),
   CONSTRAINT uq_tipo_equipo_nombre UNIQUE (nombre_tipo)
);
COMMENT ON TABLE tipo_equipo IS 'Catalogo de tipos de equipo: notebook, PC escritorio, monitor, impresora, otro';

CREATE TABLE equipo (
   equipo_id        NUMBER         GENERATED BY DEFAULT AS IDENTITY,
   cliente_id       NUMBER         NOT NULL,
   tipo_equipo_id   NUMBER         NOT NULL,
   marca            VARCHAR2(60)   NOT NULL,
   modelo           VARCHAR2(60)   NOT NULL,
   numero_serie     VARCHAR2(60),
   fecha_registro   DATE           DEFAULT SYSDATE NOT NULL,
   CONSTRAINT pk_equipo PRIMARY KEY (equipo_id),
   CONSTRAINT uq_equipo_serie UNIQUE (numero_serie),
   CONSTRAINT fk_equipo_cliente FOREIGN KEY (cliente_id)
      REFERENCES cliente (cliente_id),
   CONSTRAINT fk_equipo_tipo FOREIGN KEY (tipo_equipo_id)
      REFERENCES tipo_equipo (tipo_equipo_id)
);
COMMENT ON TABLE equipo IS 'Equipo tecnologico de un cliente que ingresa a servicio tecnico';


CREATE TABLE estado_orden (
   estado_id         NUMBER         GENERATED BY DEFAULT AS IDENTITY,
   nombre_estado     VARCHAR2(40)   NOT NULL,
   orden_secuencia   NUMBER(2)      NOT NULL,
   es_estado_final   CHAR(1)        DEFAULT 'N' NOT NULL,
   CONSTRAINT pk_estado_orden PRIMARY KEY (estado_id),
   CONSTRAINT uq_estado_orden_nombre UNIQUE (nombre_estado),
   CONSTRAINT ck_estado_final CHECK (es_estado_final IN ('S','N'))
);
COMMENT ON TABLE estado_orden IS 'Catalogo de estados del ciclo de vida de una orden de servicio';


CREATE TABLE tipo_falla (
   tipo_falla_id   NUMBER         GENERATED BY DEFAULT AS IDENTITY,
   nombre_falla    VARCHAR2(80)   NOT NULL,
   descripcion     VARCHAR2(200),
   CONSTRAINT pk_tipo_falla PRIMARY KEY (tipo_falla_id),
   CONSTRAINT uq_tipo_falla_nombre UNIQUE (nombre_falla)
);
COMMENT ON TABLE tipo_falla IS 'Catalogo de tipos de falla detectados en los equipos';


CREATE TABLE repuesto (
   repuesto_id       NUMBER          GENERATED BY DEFAULT AS IDENTITY,
   nombre_repuesto   VARCHAR2(100)   NOT NULL,
   descripcion       VARCHAR2(200),
   stock_actual      NUMBER(8)       DEFAULT 0 NOT NULL,
   stock_minimo      NUMBER(8)       DEFAULT 0 NOT NULL,
   precio_unitario   NUMBER(10,2)    DEFAULT 0,
   unidad_medida     VARCHAR2(20)    DEFAULT 'UNIDAD',
   CONSTRAINT pk_repuesto PRIMARY KEY (repuesto_id),
   CONSTRAINT ck_repuesto_stock CHECK (stock_actual >= 0)
);
COMMENT ON TABLE repuesto IS 'Catalogo de repuestos y su stock disponible';


CREATE TABLE orden_servicio (
   orden_id                    NUMBER         GENERATED BY DEFAULT AS IDENTITY,
   cliente_id                  NUMBER         NOT NULL,
   equipo_id                   NUMBER         NOT NULL,
   sucursal_id                 NUMBER         NOT NULL,
   tecnico_id                  NUMBER,
   estado_id                   NUMBER         NOT NULL,
   descripcion_problema        VARCHAR2(1000) NOT NULL,
   fecha_ingreso                DATE           DEFAULT SYSDATE NOT NULL,
   fecha_diagnostico_estimada   DATE,
   fecha_entrega_estimada       DATE,
   fecha_entrega_real           DATE,
   CONSTRAINT pk_orden_servicio PRIMARY KEY (orden_id),
   CONSTRAINT fk_orden_cliente FOREIGN KEY (cliente_id)
      REFERENCES cliente (cliente_id),
   CONSTRAINT fk_orden_equipo FOREIGN KEY (equipo_id)
      REFERENCES equipo (equipo_id),
   CONSTRAINT fk_orden_sucursal FOREIGN KEY (sucursal_id)
      REFERENCES sucursal (sucursal_id),
   CONSTRAINT fk_orden_tecnico FOREIGN KEY (tecnico_id)
      REFERENCES tecnico (tecnico_id),
   CONSTRAINT fk_orden_estado FOREIGN KEY (estado_id)
      REFERENCES estado_orden (estado_id),
   CONSTRAINT ck_orden_fechas CHECK (
      fecha_entrega_real IS NULL OR fecha_entrega_real >= fecha_ingreso
   )
);
COMMENT ON TABLE orden_servicio IS 'Orden de servicio registrada para un equipo de un cliente';


CREATE TABLE diagnostico (
   diagnostico_id                NUMBER          GENERATED BY DEFAULT AS IDENTITY,
   orden_id                      NUMBER          NOT NULL,
   tecnico_id                    NUMBER          NOT NULL,
   tipo_falla_id                 NUMBER          NOT NULL,
   fecha_diagnostico              DATE            DEFAULT SYSDATE NOT NULL,
   descripcion_diagnostico        VARCHAR2(1000)  NOT NULL,
   requiere_aprobacion_cliente    CHAR(1)         DEFAULT 'N' NOT NULL,
   CONSTRAINT pk_diagnostico PRIMARY KEY (diagnostico_id),
   CONSTRAINT fk_diag_orden FOREIGN KEY (orden_id)
      REFERENCES orden_servicio (orden_id),
   CONSTRAINT fk_diag_tecnico FOREIGN KEY (tecnico_id)
      REFERENCES tecnico (tecnico_id),
   CONSTRAINT fk_diag_falla FOREIGN KEY (tipo_falla_id)
      REFERENCES tipo_falla (tipo_falla_id),
   CONSTRAINT ck_diag_aprobacion CHECK (requiere_aprobacion_cliente IN ('S','N'))
);
COMMENT ON TABLE diagnostico IS 'Diagnostico realizado por un tecnico sobre una orden de servicio';


CREATE TABLE reparacion (
   reparacion_id         NUMBER          GENERATED BY DEFAULT AS IDENTITY,
   orden_id              NUMBER          NOT NULL,
   tecnico_id            NUMBER          NOT NULL,
   fecha_inicio          DATE            DEFAULT SYSDATE NOT NULL,
   fecha_termino         DATE,
   resultado_reparacion  VARCHAR2(20)    DEFAULT 'PENDIENTE' NOT NULL,
   observaciones         VARCHAR2(1000),
   CONSTRAINT pk_reparacion PRIMARY KEY (reparacion_id),
   CONSTRAINT fk_reparacion_orden FOREIGN KEY (orden_id)
      REFERENCES orden_servicio (orden_id),
   CONSTRAINT fk_reparacion_tecnico FOREIGN KEY (tecnico_id)
      REFERENCES tecnico (tecnico_id),
   CONSTRAINT ck_reparacion_resultado CHECK (
      resultado_reparacion IN ('PENDIENTE','EXITOSA','RECHAZADA')
   ),
   CONSTRAINT ck_reparacion_fechas CHECK (
      fecha_termino IS NULL OR fecha_termino >= fecha_inicio
   )
);
COMMENT ON TABLE reparacion IS 'Reparacion efectuada sobre una orden de servicio';

CREATE TABLE reparacion_repuesto (
   reparacion_id       NUMBER    NOT NULL,
   repuesto_id         NUMBER    NOT NULL,
   cantidad_utilizada  NUMBER(6) NOT NULL,
   CONSTRAINT pk_reparacion_repuesto PRIMARY KEY (reparacion_id, repuesto_id),
   CONSTRAINT fk_reprep_reparacion FOREIGN KEY (reparacion_id)
      REFERENCES reparacion (reparacion_id),
   CONSTRAINT fk_reprep_repuesto FOREIGN KEY (repuesto_id)
      REFERENCES repuesto (repuesto_id),
   CONSTRAINT ck_reprep_cantidad CHECK (cantidad_utilizada > 0)
);
COMMENT ON TABLE reparacion_repuesto IS 'Repuestos utilizados en cada reparacion y su cantidad';



CREATE TABLE historial_estado_orden (
   historial_id          NUMBER          GENERATED BY DEFAULT AS IDENTITY,
   orden_id              NUMBER          NOT NULL,
   estado_anterior_id    NUMBER,
   estado_nuevo_id       NUMBER          NOT NULL,
   fecha_cambio          TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,
   usuario_responsable   VARCHAR2(60),
   observacion           VARCHAR2(500),
   CONSTRAINT pk_historial_estado PRIMARY KEY (historial_id),
   CONSTRAINT fk_hist_orden FOREIGN KEY (orden_id)
      REFERENCES orden_servicio (orden_id),
   CONSTRAINT fk_hist_estado_ant FOREIGN KEY (estado_anterior_id)
      REFERENCES estado_orden (estado_id),
   CONSTRAINT fk_hist_estado_nvo FOREIGN KEY (estado_nuevo_id)
      REFERENCES estado_orden (estado_id)
);
COMMENT ON TABLE historial_estado_orden IS 'Historico de cambios de estado de cada orden de servicio';


CREATE TABLE auditoria_cambios (
   auditoria_id           NUMBER          GENERATED BY DEFAULT AS IDENTITY,
   tabla_afectada         VARCHAR2(60)    NOT NULL,
   registro_id            NUMBER          NOT NULL,
   campo_modificado       VARCHAR2(60),
   valor_anterior         VARCHAR2(500),
   valor_nuevo            VARCHAR2(500),
   operacion              VARCHAR2(10)    NOT NULL,
   fecha_modificacion     TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,
   usuario_responsable    VARCHAR2(60),
   CONSTRAINT pk_auditoria_cambios PRIMARY KEY (auditoria_id),
   CONSTRAINT ck_auditoria_operacion CHECK (
      operacion IN ('INSERT','UPDATE','DELETE')
   )
);
COMMENT ON TABLE auditoria_cambios IS
   'Registro general de trazabilidad: que cambio, quien y cuando (pensada para poblarse via triggers)';



CREATE TABLE resumen_reparacion_tecnico (
   resumen_id            NUMBER         GENERATED BY DEFAULT AS IDENTITY,
   tecnico_id            NUMBER         NOT NULL,
   tecnico_nombre        VARCHAR2(120)  NOT NULL,
   orden_id              NUMBER         NOT NULL,
   reparacion_id         NUMBER         NOT NULL,
   cliente_id            NUMBER         NOT NULL,
   cliente_nombre        VARCHAR2(150),
   cantidad_repuestos    NUMBER,
   valor_repuestos       NUMBER,
   descuento_volumen     NUMBER,
   descuento_antiguedad  NUMBER,
   descuento_total       NUMBER,
   valor_final           NUMBER,
   fecha_proceso         DATE           DEFAULT SYSDATE,
   CONSTRAINT pk_resumen_rep_tec PRIMARY KEY (resumen_id)
);
COMMENT ON TABLE resumen_reparacion_tecnico IS
   'Tabla de salida del informe PL/SQL de reparaciones por tecnico (caso C)';


COMMIT;
